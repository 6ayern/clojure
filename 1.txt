;; ------------------------------------------------------------
;; 01. Equalities
;; ------------------------------------------------------------
(ns koans.01-equalities)

(assert (= true true))
(assert (= 2 (+ 1 1)))
(assert (= (+ 3 4) 7 (+ 2 5)))
(assert (= true (= 2 2/1)))
(assert (= false (= 2 2.0)))
(assert (= true (== 2.0 2)))
(assert (= true (not (= 1 nil))))
(assert (= false (= "" :hello 'hello)))
(assert (= :hello (keyword "hello")))
(assert (= 'hello (symbol "hello")))
(assert (= nil nil))
(assert (not= 42 3))

;; ------------------------------------------------------------
;; 02. Strings
;; ------------------------------------------------------------
(ns koans.02-strings
  (:require [clojure.string :as string]))

(assert (= "" ""))
(assert (= "world" (str 'world)))
(assert (= "Cool right?" (str "Cool " "right?")))
(assert (= \C (get "Characters" 0)))
(assert (= 11 (count "Hello World")))
(assert (= false (= \c "c")))
(assert (= "World" (subs "Hello World" 6 11)))
(assert (= "123" (string/join '(1 2 3))))
(assert (= "1, 2, 3" (string/join ", " '(1 2 3))))
(assert (= ["1" "2" "3"] (string/split-lines "1\n2\n3")))
(assert (= "olleh" (string/reverse "hello")))
(assert (= 0 (string/index-of "hello world" "hello")))
(assert (= 13 (string/last-index-of "hello world, hello" "hello")))
(assert (= nil (string/index-of "hello world" "bob")))
(assert (= "hello world" (string/trim "  \nhello world \t \n")))
(assert (= true (char? \c)))
(assert (= false (char? "a")))
(assert (= false (string? \b)))
(assert (= true (string? "a")))
(assert (= true (string/blank? "")))
(assert (= true (string/blank? " \n \t  ")))
(assert (= false (string/blank? "hello?\nare you out there?")))

(ns koans.03-lists)

(assert (= '(1 2 3 4 5) (list 1 2 3 4 5)))
(assert (= 1 (first '(1 2 3 4 5))))
(assert (= '(2 3 4 5) (rest '(1 2 3 4 5))))
(assert (= 3 (count '(dracula dooku chocula))))
(assert (= 0 (count '())))
(assert (= () (rest '(100))))
(assert (= '(:a :b :c :d :e) (cons :a '(:b :c :d :e))))
(assert (= '(:e :a :b :c :d) (conj '(:a :b :c :d) :e)))
(assert (= :a (peek '(:a :b :c :d :e))))
(assert (= '(:b :c :d :e) (pop '(:a :b :c :d :e))))
(assert (= "No dice!" (try
                        (pop '())
                        (catch IllegalStateException e
                          "No dice!"))))
(assert (= () (try
                (rest '())
                (catch IllegalStateException e
                  "No dice!"))))

;; 06. Maps
;; ------------------------------------------------------------
(ns koans.06-maps)

(assert (= {:a 1 :b 2} (hash-map :a 1 :b 2)))
(assert (= {:a 1} (hash-map :a 1)))
(assert (= 2 (count {:a 1 :b 2})))
(assert (= 2 (get {:a 1 :b 2} :b)))
(assert (= 1 ({:a 1 :b 2} :a)))
(assert (= 1 (:a {:a 1 :b 2})))
(assert (= "Sochi" ({2010 "Vancouver" 2014 "Sochi" 2018 "PyeongChang"} 2014)))
(assert (= nil (get {:a 1 :b 2} :c)))
(assert (= :key-not-found (get {:a 1 :b 2} :c :key-not-found)))
(assert (= true (contains? {:a nil :b nil} :b)))
(assert (= false (contains? {:a nil :b nil} :c)))
(assert (= {1 "January" 2 "February"} (assoc {1 "January"} 2 "February")))
(assert (= {1 "January"} (dissoc {1 "January" 2 "February"} 2)))
(assert (= {:a 1 :b 2 :c 3} (merge {:a 1 :b 2} {:c 3})))
(assert (= {:a 1 :b 2 :c 3} (merge-with + {:a 1 :b 1} {:b 1 :c 3})))
(assert (= (list 2010 2014 2018)
           (sort (keys {2014 "Sochi" 2018 "PyeongChang" 2010 "Vancouver"}))))
(assert (= (list "PyeongChang" "Sochi" "Vancouver")
           (sort (vals {2010 "Vancouver" 2014 "Sochi" 2018 "PyeongChang"}))))
(assert (= {:a 2 :b 3}
           (into {}
                 (map
                  (fn [[k v]] [k (inc v)])
                  {:a 1 :b 2}))))

;; 08. Conditionals

(ns koans.08-conditionals)

(defn explain-exercise-velocity [exercise-term]
  (case exercise-term
        :bicycling        "pretty fast"
        :jogging          "not super fast"
        :walking          "not fast at all"
        "is that even exercise?"))

(assert (= :a (if (false? (= 4 5)) :a :b)))
(assert (= [] (if (> 4 3) [])))
(assert (= nil (if (nil? 0) [:a :b :c])))
(assert (= :glory (if (not (empty? ())) :doom :glory)))
(let [x 5]
  (assert (= :your-road (cond (= x 4) :road-not-taken
                              (= x 6) :another-road-not-taken
                              :else :your-road))))
(assert (= 'doom (if-not (zero? 1) 'doom 'more-doom)))
(assert (= "pretty fast" (explain-exercise-velocity :bicycling)))
(assert (= "is that even exercise?" (explain-exercise-velocity :watching-tv)))

(ns koans.09-higher-order-functions)

(assert (= [4 8 12] (map (fn [x] (* 4 x)) [1 2 3])))
(assert (= [1 4 9 16 25] (map (fn [x] (* x x)) [1 2 3 4 5])))
(assert (= [false false true false false] (map nil? [:a :b nil :c :d])))

;; ------------------------------------------------------------
;; 10. Runtime Polymorphism
;; ------------------------------------------------------------
(ns koans.10-runtime-polymorphism)

(defn hello
  ([] "Hello World!")
  ([a] (str "Hello, you silly " a "."))
  ([a & more] (str "Hello to this group: " (apply str (interpose ", " (cons a more))) "!")))

(defmulti diet (fn [x] (:eater x)))
(defmethod diet :herbivore [a] (str (:name a) " eats veggies."))
(defmethod diet :carnivore [a] (str (:name a) " eats animals."))
(defmethod diet :default [a] (str "I don't know what " (:name a) " eats."))

(assert (= "Hello World!" (hello)))
(assert (= "Hello, you silly world." (hello "world")))
(assert (= "Hello to this group: Peter, Paul, Mary!" (hello "Peter" "Paul" "Mary")))
(assert (= "Bambi eats veggies." (diet {:species "deer" :name "Bambi" :age 1 :eater :herbivore})))
(assert (= "Thumper eats veggies." (diet {:species "rabbit" :name "Thumper" :age 1 :eater :herbivore})))
(assert (= "Simba eats animals." (diet {:species "lion" :name "Simba" :age 1 :eater :carnivore})))
(assert (= "I don't know what Rich Hickey eats." (diet {:name "Rich Hickey"})))

(ns koans.13-creating-functions)

(defn square [x] (* x x))

;; ------------------------------------------------------------
;; 14. Recursion
;; ------------------------------------------------------------
(ns koans.14-recursion)

(defn is-even? [n] (if (= n 0) true (not (is-even? (dec n)))))
(defn is-even-bigint? [n]
  (loop [n n acc true]
    (if (= n 0) acc (recur (dec n) (not acc)))))
(defn recursive-reverse [coll]
  (loop [coll coll acc ()]
    (if (seq coll) (recur (rest coll) (conj acc (first coll))) acc)))
(defn factorial [n]
  (loop [n n acc 1]
    (if (zero? n) acc (recur (dec n) (* acc n)))))

(assert (= true (is-even? 0)))
(assert (= false (is-even? 1)))
(assert (= false (is-even-bigint? 100003N)))
(assert (= '(1) (recursive-reverse [1])))
(assert (= '(6 5 4 3 2) (recursive-reverse [2 3 4 5 6])))
(assert (= 1 (factorial 1)))
(assert (= 2 (factorial 2)))
(assert (= 6 (factorial 3)))
(assert (= 24 (factorial 4)))
(assert (< 1000000000000000000000000N (factorial 1000N)))
(assert (< 1000000000000000000000000N (factorial 100003N)))

;; 15. Destructuring
;; ------------------------------------------------------------
(ns koans.15-destructuring)

(def test-address {:street-address "123 Test Lane" :city "Testerville" :state "TX"})

(assert (= ":bar:foo" ((fn [[a b]] (str b a)) [:foo :bar])))
(assert (= (str "An Oxford comma list of apples, " "oranges, " "and pears.")
           ((fn [[a b c]] (format (str "An Oxford comma list of %s, " "%s, " "and %s.") a b c))
            ["apples" "oranges" "pears"])))
(assert (= "Rich Hickey aka The Clojurer aka Go Time aka Lambda Guru"
           (let [[first-name last-name & aliases] (list "Rich" "Hickey" "The Clojurer" "Go Time" "Lambda Guru")]
             (apply str (interpose " " (apply list first-name last-name (interleave (repeat "aka") aliases)))))))
(assert (= {:original-parts ["Stephen" "Hawking"] :named-parts {:first "Stephen" :last "Hawking"}}
           (let [[first-name last-name :as full-name] ["Stephen" "Hawking"]]
             {:original-parts full-name :named-parts {:first first-name :last last-name}})))
(assert (= "123 Test Lane, Testerville, TX"
           (let [{street-address :street-address, city :city, state :state} test-address]
             (str street-address ", " city ", " state))))
(assert (= "123 Test Lane, Testerville, TX"
           (let [{:keys [street-address city state]} test-address]
             (str street-address ", " city ", " state))))
(assert (= "Test Testerson, 123 Test Lane, Testerville, TX"
           ((fn [[fname lname] {:keys [street-address city state]}]
              (str fname " " lname ", " street-address ", " city ", " state)) ["Test" "Testerson"] test-address)))

;; 16. Refs
;; ------------------------------------------------------------
(ns koans.16-refs)

(def the-world (ref "hello"))
(def bizarro-world (ref {}))

(assert (= "hello" (deref the-world)))
(assert (= "hello" @the-world))
(assert (= "better" (do (dosync (ref-set the-world "better")) @the-world)))
(assert (= "better!!!" (let [exclamator (fn [x] (str x "!"))]
                         (dosync (alter the-world exclamator) (alter the-world exclamator) (alter the-world exclamator))
                         @the-world)))
(assert (= 0 (do (dosync (ref-set the-world 0)) @the-world)))
(assert (= 20 (do (dosync (alter the-world (fn [x] (+ 20 x)))))))
(assert (= ["Real Jerry" "Bizarro Jerry"]
           (do (dosync
                (ref-set the-world {})
                (alter the-world assoc :jerry "Real Jerry")
                (alter bizarro-world assoc :jerry "Bizarro Jerry")
                (map :jerry [@the-world @bizarro-world])))))
